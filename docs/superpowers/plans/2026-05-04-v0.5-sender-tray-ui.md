# v0.5 Sender Tray UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the CLI sender with a system-tray application that provides a Popover panel for device discovery, PIN pairing, and streaming control using egui.

**Architecture:** A single eframe window (initially hidden) with a tray-icon. The main thread runs the egui event loop + tray event processing. Background threads handle mDNS browsing and the streaming engine, communicating via mpsc channels. State is modeled as a Rust enum (Idle → Connecting → ModeSelect → Streaming → Paused).

**Tech Stack:** `eframe` 0.34+, `egui` 0.34+, `tray-icon` 0.19+, `image` 0.25+

Create `src/ui/tray.rs`:

```rust
use tray_icon::icon::Icon;
use tray_icon::menu::Menu;
use tray_icon::{TrayIcon, TrayIconBuilder};

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum TrayState {
    Idle,
    Streaming,
    Paused,
}

pub struct AppTray {
    tray: TrayIcon,
    state: TrayState,
    icon_idle: Icon,
    icon_streaming: Icon,
    icon_paused: Icon,
}

fn create_icon(r: u8, g: u8, b: u8) -> Icon {
    let size = 22u32;
    let mut rgba = vec![0u8; (size * size * 4) as usize];
    let center = size as f32 / 2.0;
    let radius = size as f32 / 2.0 - 1.0;
    for y in 0..size {
        for x in 0..size {
            let dx = x as f32 - center;
            let dy = y as f32 - center;
            let dist = (dx * dx + dy * dy).sqrt();
            let idx = ((y * size + x) * 4) as usize;
            if dist <= radius {
                rgba[idx] = r;
                rgba[idx + 1] = g;
                rgba[idx + 2] = b;
                rgba[idx + 3] = 255;
            }
        }
    }
    Icon::from_rgba(rgba, size, size).expect("valid icon data")
}

impl AppTray {
    pub fn new() -> Self {
        let icon_idle = create_icon(128, 128, 128);
        let icon_streaming = create_icon(39, 174, 96);
        let icon_paused = create_icon(243, 156, 18);

        let tray = TrayIconBuilder::new()
            .with_icon(icon_idle.clone())
            .with_tooltip("screen-mirror")
            .build()
            .expect("failed to create tray icon");

        Self {
            tray,
            state: TrayState::Idle,
            icon_idle,
            icon_streaming,
            icon_paused,
        }
    }

    pub fn set_state(&mut self, state: TrayState) {
        if self.state == state {
            return;
        }
        self.state = state;
        let icon = match state {
            TrayState::Idle => &self.icon_idle,
            TrayState::Streaming => &self.icon_streaming,
            TrayState::Paused => &self.icon_paused,
        };
        let _ = self.tray.set_icon(Some(icon.clone()));
    }
}

#[cfg(target_os = "macos")]
pub fn calculate_position(icon_rect: tray_icon::Rect) -> egui::Pos2 {
    use crate::ui::theme::PANEL_WIDTH;
    let x = icon_rect.position.x + icon_rect.size.width / 2.0 - PANEL_WIDTH as f64 / 2.0;
    let y = icon_rect.position.y + icon_rect.size.height;
    egui::pos2(x as f32, y as f32)
}

#[cfg(target_os = "windows")]
pub fn calculate_position(icon_rect: tray_icon::Rect) -> egui::Pos2 {
    use crate::ui::theme::{PANEL_MAX_HEIGHT, PANEL_WIDTH};
    let x = icon_rect.position.x + icon_rect.size.width / 2.0 - PANEL_WIDTH as f64 / 2.0;
    let y = icon_rect.position.y - PANEL_MAX_HEIGHT as f64;
    egui::pos2(x as f32, y as f32)
}

#[cfg(not(any(target_os = "macos", target_os = "windows")))]
pub fn calculate_position(icon_rect: tray_icon::Rect) -> egui::Pos2 {
    use crate::ui::theme::PANEL_WIDTH;
    let x = icon_rect.position.x + icon_rect.size.width / 2.0 - PANEL_WIDTH as f64 / 2.0;
    let y = icon_rect.position.y + icon_rect.size.height;
    egui::pos2(x as f32, y as f32)
}
```

```rust
use egui::{Ui, RichText, TextEdit, Button, Vec2, Color32, Sense};

use crate::discovery::browser::DiscoveredReceiver;
use crate::ui::theme::*;

pub struct IdleViewState {
    pub devices: Vec<DiscoveredReceiver>,
    pub selected_device: Option<usize>,
    pub pin_input: String,
    pub error: Option<String>,
    pub connecting: bool,
    pub connecting_device: Option<String>,
}

pub enum IdleAction {
    None,
    Connect { device_index: usize, pin: String },
}

pub fn render(ui: &mut Ui, state: &mut IdleViewState) -> IdleAction {
    let mut action = IdleAction::None;

    ui.add_space(PADDING);
    ui.horizontal(|ui| {
        ui.add_space(PADDING);
        ui.label(RichText::new("screen-mirror").strong().size(14.0));
    });
    ui.add_space(SPACING);
    ui.separator();
    ui.add_space(SPACING);

    if state.connecting {
        ui.horizontal(|ui| {
            ui.add_space(PADDING);
            ui.spinner();
            ui.label(format!(
                "正在连接 {}...",
                state.connecting_device.as_deref().unwrap_or("")
            ));
        });
        ui.add_space(PADDING);
        return action;
    }

    // Device discovery status
    ui.horizontal(|ui| {
        ui.add_space(PADDING);
        if state.devices.is_empty() {
            ui.label(RichText::new("正在搜索设备...").color(COLOR_MUTED));
        } else {
            ui.label(RichText::new(format!("发现 {} 台设备", state.devices.len())).color(COLOR_MUTED));
        }
    });
    ui.add_space(SPACING);

    // Device list
    egui::ScrollArea::vertical()
        .max_height(150.0)
        .show(ui, |ui| {
            for (idx, device) in state.devices.iter().enumerate() {
                let selected = state.selected_device == Some(idx);
                let response = ui
                    .horizontal(|ui| {
                        ui.add_space(PADDING);
                        let bg = if selected { COLOR_ACCENT.linear_multiply(0.2) } else { Color32::TRANSPARENT };
                        let rect = ui.available_rect_before_wrap();
                        if selected {
                            ui.painter().rect_filled(rect, 4.0, bg);
                        }
                        ui.label("📺");
                        ui.vertical(|ui| {
                            ui.label(RichText::new(&device.name).strong());
                            ui.label(RichText::new(device.addr.to_string()).small().color(COLOR_MUTED));
                        });
                    })
                    .response;
                if response.interact(Sense::click()).clicked() {
                    state.selected_device = Some(idx);
                }
                ui.add_space(2.0);
            }
        });

    ui.add_space(SPACING);
    ui.separator();
    ui.add_space(SPACING);

    // PIN input
    ui.horizontal(|ui| {
        ui.add_space(PADDING);
        ui.label("投屏码:");
        let response = ui.add(
            TextEdit::singleline(&mut state.pin_input)
                .desired_width(100.0)
                .font(egui::TextStyle::Monospace)
                .hint_text("6位数字")
        );
        // Filter non-digits and limit to 6
        state.pin_input.retain(|c| c.is_ascii_digit());
        if state.pin_input.len() > 6 {
            state.pin_input.truncate(6);
        }
    });

    ui.add_space(SPACING);

    // Connect button
    let can_connect = state.selected_device.is_some() && state.pin_input.len() == 6;
    ui.horizontal(|ui| {
        ui.add_space(PADDING);
        let btn = Button::new("连 接")
            .min_size(Vec2::new(ui.available_width() - PADDING * 2.0, BUTTON_HEIGHT));
        if ui.add_enabled(can_connect, btn).clicked() {
            if let Some(idx) = state.selected_device {
                action = IdleAction::Connect {
                    device_index: idx,
                    pin: state.pin_input.clone(),
                };
            }
        }
    });

    // Error display
    if let Some(error) = &state.error {
        ui.add_space(SPACING);
        ui.horizontal(|ui| {
            ui.add_space(PADDING);
            ui.label(RichText::new(format!("❌ {error}")).color(COLOR_ERROR));
        });
    }

    ui.add_space(PADDING);
    action
}
```

```rust
fn format_bitrate(bps: u64) -> String {
    if bps >= 1_000_000 {
        format!("{:.1} Mbps", bps as f64 / 1_000_000.0)
    } else if bps >= 1_000 {
        format!("{:.0} Kbps", bps as f64 / 1_000.0)

        format!("{bps} bps")
```

```rust
use std::sync::mpsc::{self, Receiver, Sender};
use std::time::Instant;

use eframe::egui;

use crate::discovery::browser::DiscoveredReceiver;
use crate::ui::messages::*;
use crate::ui::theme::*;
use crate::ui::tray::{self, AppTray, TrayState};
use crate::ui::views::{idle, mode, streaming, paused};

pub enum AppState {
    Idle,
    Connecting { device_name: String, started: Instant },
    ModeSelect { device_name: String },
    Streaming { device_name: String, stats: StreamStats },
    Paused { device_name: String, stats: StreamStats },
}

pub struct App {
    state: AppState,
    visible: bool,
    tray: Option<AppTray>,
    // Channel: UI sends commands to backend
    cmd_tx: Sender<UiCommand>,
    // Channel: backend sends events to UI
    event_rx: Receiver<BackendEvent>,
    // Idle view state (persists across show/hide)
    idle_state: idle::IdleViewState,
}

impl App {
    pub fn new(
        cmd_tx: Sender<UiCommand>,
        event_rx: Receiver<BackendEvent>,
    ) -> Self {
        Self {
            state: AppState::Idle,
            visible: false,
            tray: None,
            cmd_tx,
            event_rx,
            idle_state: idle::IdleViewState {
                devices: Vec::new(),
                selected_device: None,
                pin_input: String::new(),
                error: None,
                connecting: false,
                connecting_device: None,
            },
        }
    }

    fn handle_backend_events(&mut self) {
        while let Ok(event) = self.event_rx.try_recv() {
            match event {
                BackendEvent::DevicesUpdated(devices) => {
                    self.idle_state.devices = devices;
                    // Reset selection if device list changed and selected is out of bounds
                    if let Some(idx) = self.idle_state.selected_device {
                        if idx >= self.idle_state.devices.len() {
                            self.idle_state.selected_device = None;
                        }
                    }
                }
                BackendEvent::PairingSuccess => {
                    if let AppState::Connecting { device_name, .. } = &self.state {
                        let name = device_name.clone();
                        self.state = AppState::ModeSelect { device_name: name };
                        self.idle_state.connecting = false;
                        self.idle_state.connecting_device = None;
                    }
                }
                BackendEvent::PairingFailed(reason) => {
                    self.idle_state.error = Some(reason);
                    self.idle_state.connecting = false;
                    self.idle_state.connecting_device = None;
                    self.state = AppState::Idle;
                }
                BackendEvent::PairingTimeout => {
                    self.idle_state.error = Some("连接超时".to_string());
                    self.idle_state.connecting = false;
                    self.idle_state.connecting_device = None;
                    self.state = AppState::Idle;
                }
                BackendEvent::StreamingStarted => {
                    if let AppState::ModeSelect { device_name } = &self.state {
                        let name = device_name.clone();
                        self.state = AppState::Streaming {
                            device_name: name,
                            stats: StreamStats {
                                resolution_w: 0,
                                resolution_h: 0,
                                fps: 0.0,
                                bitrate_bps: 0,
                                latency_ms: 0.0,
                                packet_loss_pct: 0.0,
                            },
                        };
                    }
                }
                BackendEvent::StatsUpdate(stats) => {
                    match &mut self.state {
                        AppState::Streaming { stats: s, .. } => *s = stats,
                        AppState::Paused { stats: s, .. } => *s = stats,
                        _ => {}
                    }
                }
                BackendEvent::Disconnected(reason) => {
                    if !reason.is_empty() {
                        self.idle_state.error = Some(reason);
                    }
                    self.state = AppState::Idle;
                    self.idle_state.connecting = false;
                    self.idle_state.connecting_device = None;
                }
            }
        }
    }

    fn update_tray_state(&mut self) {
        if let Some(tray) = &mut self.tray {
            let tray_state = match &self.state {
                AppState::Idle | AppState::Connecting { .. } | AppState::ModeSelect { .. } => TrayState::Idle,
                AppState::Streaming { .. } => TrayState::Streaming,
                AppState::Paused { .. } => TrayState::Paused,
            };
            tray.set_state(tray_state);
        }
    }

    fn check_connecting_timeout(&mut self) {
        if let AppState::Connecting { started, .. } = &self.state {
            if started.elapsed().as_secs() >= 10 {
                self.idle_state.error = Some("连接超时".to_string());
                self.idle_state.connecting = false;
                self.idle_state.connecting_device = None;
                self.state = AppState::Idle;
            }
        }
    }

    fn render_current_state(&mut self, ctx: &egui::Context) {
        egui::CentralPanel::default().show(ctx, |ui| {
            match &self.state {
                AppState::Idle | AppState::Connecting { .. } => {
                    let action = idle::render(ui, &mut self.idle_state);
                    match action {
                        idle::IdleAction::Connect { device_index, pin } => {
                            if let Some(device) = self.idle_state.devices.get(device_index) {
                                let addr = device.addr;
                                let name = device.name.clone();
                                self.idle_state.connecting = true;
                                self.idle_state.connecting_device = Some(name.clone());
                                self.idle_state.error = None;
                                self.idle_state.pin_input.clear();
                                self.state = AppState::Connecting {
                                    device_name: name,
                                    started: Instant::now(),
                                };
                                let _ = self.cmd_tx.send(UiCommand::Connect { addr, pin });
                            }
                        }
                        idle::IdleAction::None => {}
                    }
                }
                AppState::ModeSelect { device_name } => {
                    let name = device_name.clone();
                    let action = mode::render(ui, &name);
                    match action {
                        mode::ModeAction::Start(mode) => {
                            let _ = self.cmd_tx.send(UiCommand::StartStreaming { mode });
                        }
                        mode::ModeAction::None => {}
                    }
                }
                AppState::Streaming { device_name, stats } => {
                    let name = device_name.clone();
                    let s = stats.clone();
                    let action = streaming::render(ui, &name, &s);
                    match action {
                        streaming::StreamingAction::Pause => {
                            let _ = self.cmd_tx.send(UiCommand::Pause);
                            if let AppState::Streaming { device_name, stats } = std::mem::replace(&mut self.state, AppState::Idle) {
                                self.state = AppState::Paused { device_name, stats };
                            }
                        }
                        streaming::StreamingAction::Disconnect => {
                            let _ = self.cmd_tx.send(UiCommand::Disconnect);
                            self.state = AppState::Idle;
                        }
                        streaming::StreamingAction::None => {}
                    }
                }
                AppState::Paused { device_name, stats } => {
                    let name = device_name.clone();
                    let action = paused::render(ui, &name);
                    match action {
                        paused::PausedAction::Resume => {
                            let _ = self.cmd_tx.send(UiCommand::Resume);
                            if let AppState::Paused { device_name, stats } = std::mem::replace(&mut self.state, AppState::Idle) {
                                self.state = AppState::Streaming { device_name, stats };
                            }
                        }
                        paused::PausedAction::Disconnect => {
                            let _ = self.cmd_tx.send(UiCommand::Disconnect);
                            self.state = AppState::Idle;
                        }
                        paused::PausedAction::None => {}
                    }
                }
            }
        });
    }
}

impl eframe::App for App {
    fn update(&mut self, ctx: &egui::Context, _frame: &mut eframe::Frame) {
        // Initialize tray on first frame (must be on main thread)
        if self.tray.is_none() {
            self.tray = Some(AppTray::new());
        }

        // Handle tray click events
        if let Ok(event) = tray_icon::TrayIconEvent::receiver().try_recv() {
            if let tray_icon::TrayIconEvent::Click { rect, .. } = event {
                self.visible = !self.visible;
                if self.visible {
                    let pos = tray::calculate_position(rect);
                    ctx.send_viewport_cmd(egui::ViewportCommand::OuterPosition(pos.to_pos2()));
                }
                ctx.send_viewport_cmd(egui::ViewportCommand::Visible(self.visible));
                if self.visible {
                    ctx.send_viewport_cmd(egui::ViewportCommand::Focus);
                }
            }
        }

        // Focus lost → hide
        if ctx.input(|i| i.viewport().focused == Some(false)) && self.visible {
            self.visible = false;
            ctx.send_viewport_cmd(egui::ViewportCommand::Visible(false));
        }

        // Process backend events
        self.handle_backend_events();
        self.check_connecting_timeout();
        self.update_tray_state();

        // Render UI only when visible
        if self.visible {
            self.render_current_state(ctx);
        }

        // Request repaint for continuous updates during streaming or connecting
        match &self.state {
            AppState::Streaming { .. } | AppState::Connecting { .. } => {
                ctx.request_repaint_after(std::time::Duration::from_millis(500));
            }
            _ => {}
        }
    }
}

/// Launch the tray application. Blocks until exit.
pub fn run(cmd_tx: Sender<UiCommand>, event_rx: Receiver<BackendEvent>) -> eframe::Result<()> {
    let options = eframe::NativeOptions {
        viewport: egui::ViewportBuilder::default()
            .with_decorations(false)
            .with_always_on_top()
            .with_transparent(true)
            .with_inner_size([PANEL_WIDTH, PANEL_MAX_HEIGHT])
            .with_visible(false),
        ..Default::default()
    };

    eframe::run_native(
        "screen-mirror",
        options,
        Box::new(move |_cc| {
            Ok(Box::new(App::new(cmd_tx, event_rx)))
        }),
    )
}
```

```rust
pub fn spawn_command_handler(
    cmd_rx: Receiver<UiCommand>,
    event_tx: Sender<BackendEvent>,
) -> std::thread::JoinHandle<()> {
    std::thread::spawn(move || {
        let mut streaming_active = Arc::new(AtomicBool::new(false));
        let mut paused = Arc::new(AtomicBool::new(false));

        while let Ok(cmd) = cmd_rx.recv() {
            match cmd {
                UiCommand::Connect { addr, pin } => {
                    match do_connect(addr, &pin) {
                        Ok(session) => {
                            if event_tx.send(BackendEvent::PairingSuccess).is_err() {
                                break;
                            }
                            // Wait for StartStreaming command
                            if let Ok(UiCommand::StartStreaming { mode }) = cmd_rx.recv() {
                                streaming_active = Arc::new(AtomicBool::new(true));
                                paused = Arc::new(AtomicBool::new(false));
                                let active = streaming_active.clone();
                                let is_paused = paused.clone();
                                let tx = event_tx.clone();

                                match do_streaming(session, active.clone(), is_paused.clone(), tx.clone()) {
                                    Ok(()) => {
                                        let _ = tx.send(BackendEvent::Disconnected(String::new()));
                                    }
                                    Err(e) => {
                                        let _ = tx.send(BackendEvent::Disconnected(e.to_string()));
                                    }
                                }
                            }
                        }
                        Err(e) => {
                            let msg = e.to_string();
                            if msg.contains("timed out") || msg.contains("timeout") {
                                let _ = event_tx.send(BackendEvent::PairingTimeout);
                            } else {
                                let _ = event_tx.send(BackendEvent::PairingFailed(msg));
                            }
                        }
                    }
                }
                UiCommand::Pause => {
                    paused.store(true, Ordering::Relaxed);
                }
                UiCommand::Resume => {
                    paused.store(false, Ordering::Relaxed);
                }
                UiCommand::Disconnect => {
                    streaming_active.store(false, Ordering::Relaxed);
                }
                UiCommand::StartStreaming { .. } => {
                    // Should not arrive here without a prior Connect
                }
            }
        }
    })
}

struct ConnectedSession {
    channel: SecureChannel,
    keys: crate::security::keys::SessionKeys,
    target_addr: std::net::SocketAddr,
}

fn do_connect(addr: std::net::SocketAddr, pin: &str) -> Result<ConnectedSession> {
    let mut stream = TcpStream::connect_timeout(&addr, Duration::from_secs(10))?;

    let (msg_a, state) = pairing::sender_start(pin);
    stream.write_all(&(msg_a.len() as u32).to_be_bytes())?;
    stream.write_all(&msg_a)?;

    stream.set_read_timeout(Some(Duration::from_secs(10)))?;

    let mut len_buf = [0u8; 4];
    std::io::Read::read_exact(&mut stream, &mut len_buf)?;
    let pb_len = u32::from_be_bytes(len_buf) as usize;
    if pb_len > 256 {
        anyhow::bail!("pairing message too large: {pb_len}");
    }
    let mut msg_b = vec![0u8; pb_len];
    std::io::Read::read_exact(&mut stream, &mut msg_b)?;

    let keys = pairing::sender_finish(state, &msg_b)?;
    stream.set_read_timeout(None)?;

    let channel = SecureChannel::new(stream, &keys.control_key, true);

    Ok(ConnectedSession {
        channel,
        keys,
        target_addr: addr,
    })
}

fn do_streaming(
    mut session: ConnectedSession,
    active: Arc<AtomicBool>,
    paused: Arc<AtomicBool>,
    event_tx: Sender<BackendEvent>,
) -> Result<()> {
    #[cfg(target_os = "macos")]
    let (cap_w, cap_h, capture) = {
        use crate::capture::macos::{MacOsCapture, native_resolution};
        let (w, h) = native_resolution();
        let capture = MacOsCapture::new(&CaptureConfig { fps: 0, width: w, height: h })?;
        (capture.width(), capture.height(), capture)
    };

    #[cfg(target_os = "windows")]
    let (cap_w, cap_h, capture, audio_capture) = {
        use crate::capture::windows::{DxgiCapture, WasapiCapture};
        use crate::capture::AudioCapture;
        let capture = DxgiCapture::new(&CaptureConfig { fps: 0, width: 0, height: 0 })?;
        let w = capture.width();
        let h = capture.height();
        let audio_capture = WasapiCapture::new(48000, 2)?;
        (w, h, capture, audio_capture)
    };

    let pixels = cap_w as u64 * cap_h as u64;
    let bitrate = ((pixels * 15_000_000 / (1920 * 1080)) as u32).min(40_000_000);
    let fps: u32 = if pixels > 1920 * 1200 { 30 } else { 60 };

    let udp_socket = UdpSocket::bind("0.0.0.0:0")?;
    let sender_udp_port = udp_socket.local_addr()?.port();

    let offer = Offer {
        video: OfferVideo {
            codec: "h264".to_string(),
            width: cap_w,
            height: cap_h,
            fps,
            bitrate,
        },
        audio: OfferAudio {
            codec: "opus".to_string(),
            sample_rate: 48000,
            channels: 2,
            bitrate: 128_000,
        },
        transport: OfferTransport {
            udp_port: sender_udp_port,
            fec_group_size: 6,
        },
    };

    session.channel.send(&NegotiateMessage::Offer(offer.clone()).to_bytes())?;
    session.channel.set_read_timeout(Some(Duration::from_secs(5)))?;
    let answer_bytes = session.channel
        .recv()
        .map_err(|e| anyhow::anyhow!("negotiation failed: {e}"))?
        .ok_or_else(|| anyhow::anyhow!("connection closed during negotiation"))?;
    let answer = match NegotiateMessage::from_bytes(&answer_bytes)? {
        NegotiateMessage::Answer(a) => a,
        _ => anyhow::bail!("expected Answer"),
    };
    let params = NegotiatedParams::resolve(&offer, &answer);
    session.channel.set_read_timeout(None)?;

    let receiver_media_addr = std::net::SocketAddr::new(session.target_addr.ip(), params.receiver_udp_port);
    udp_socket.connect(receiver_media_addr)?;
    let mut media_cipher = Cipher::new(&session.keys.media_key, [0, 0, 0, 1]);

    #[cfg(target_os = "macos")]
    let mut encoder = VideoEncoder::new(&EncoderConfig {
        width: params.video_width,
        height: params.video_height,
        fps: params.video_fps,
        bitrate: params.video_bitrate,
    })?;

    #[cfg(target_os = "windows")]
    let mut encoder = VideoEncoder::new_with_device(&EncoderConfig {
        width: params.video_width,
        height: params.video_height,
        fps: params.video_fps,
        bitrate: params.video_bitrate,
    }, capture.device())?;

    let mut packetizer = H264Packetizer::new(VIDEO_PT, VIDEO_SSRC, 1400);
    let mut video_fec = FecEncoder::new(params.fec_group_size, VIDEO_PT);

    let audio_config = AudioConfig {
        sample_rate: 48000,
        channels: 2,
        bitrate: 128_000,
        frame_duration_ms: 10,
    };
    let mut opus_encoder = OpusEncoder::new(&audio_config)?;
    let frame_size_samples = opus_encoder.frame_size_samples();
    let mut audio_pcm_buf: Vec<f32> = Vec::with_capacity(frame_size_samples * 2);
    let mut audio_seq: u16 = 0;
    let mut audio_ts: u32 = 0;
    let mut audio_fec = FecEncoder::new(params.fec_group_size, AUDIO_PT);

    let mut frame_count: u64 = 0;
    let mut last_stats_time = std::time::Instant::now();

    let _ = event_tx.send(BackendEvent::StreamingStarted);

    loop {
        if !active.load(Ordering::Relaxed) {
            break;
        }

        if paused.load(Ordering::Relaxed) {
            std::thread::sleep(Duration::from_millis(50));
            continue;
        }

        let Some(frame) = capture.next_frame() else {
            continue;
        };

        encoder.encode(frame)?;
        frame_count += 1;

        while let Some(encoded) = encoder.next_encoded() {
            let rtp_ts = (encoded.timestamp / 11_111) as u32;
            let nal_count = encoded.nal_units.len();
            for (i, nal) in encoded.nal_units.iter().enumerate() {
                let rtp_packets = packetizer.packetize(nal, rtp_ts);
                for pkt in &rtp_packets {
                    let encrypted = media_cipher.seal(&pkt.serialize());
                    udp_socket.send(&encrypted)?;
                    if let Some(fec_pkt) = video_fec.push(pkt) {
                        let encrypted_fec = media_cipher.seal(&fec_pkt.serialize());
                        udp_socket.send(&encrypted_fec)?;
                    }
                }
                if i == nal_count - 1 {
                    if let Some(fec_pkt) = video_fec.flush() {
                        let encrypted_fec = media_cipher.seal(&fec_pkt.serialize());
                        udp_socket.send(&encrypted_fec)?;
                    }
                }
            }
        }

        #[cfg(target_os = "macos")]
        let audio_iter = std::iter::from_fn(|| capture.try_next_audio());
        #[cfg(target_os = "windows")]
        let audio_iter = std::iter::from_fn(|| audio_capture.try_next_audio());

        for pcm in audio_iter {
            audio_pcm_buf.extend_from_slice(&pcm);
            while audio_pcm_buf.len() >= frame_size_samples {
                let frame_pcm: Vec<f32> = audio_pcm_buf.drain(..frame_size_samples).collect();
                let opus_data = opus_encoder.encode(&frame_pcm)?;

                let audio_pkt = RtpPacket {
                    header: RtpHeader {
                        version: 2,
                        padding: false,
                        extension: false,
                        marker: false,
                        payload_type: AUDIO_PT,
                        sequence_number: audio_seq,
                        timestamp: audio_ts,
                        ssrc: AUDIO_SSRC,
                    },
                    payload: opus_data.to_vec(),
                };
                audio_seq = audio_seq.wrapping_add(1);
                audio_ts = audio_ts.wrapping_add(480);

                let encrypted = media_cipher.seal(&audio_pkt.serialize());
                udp_socket.send(&encrypted)?;
                if let Some(fec_pkt) = audio_fec.push(&audio_pkt) {
                    let encrypted_fec = media_cipher.seal(&fec_pkt.serialize());
                    udp_socket.send(&encrypted_fec)?;
                }
            }
        }

        // Send stats every 500ms
        if last_stats_time.elapsed() >= Duration::from_millis(500) {
            let elapsed = last_stats_time.elapsed().as_secs_f32();
            let _ = event_tx.send(BackendEvent::StatsUpdate(StreamStats {
                resolution_w: params.video_width,
                resolution_h: params.video_height,
                fps: frame_count as f32 / elapsed,
                bitrate_bps: params.video_bitrate as u64,
                latency_ms: 0.0, // TODO: measure actual RTT in future version
                packet_loss_pct: 0.0, // TODO: measure from RTCP feedback
            }));
            frame_count = 0;
            last_stats_time = std::time::Instant::now();
        }
    }

    let _ = session.channel.shutdown();
    Ok(())
}
```
