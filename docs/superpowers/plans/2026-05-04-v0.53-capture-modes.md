> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement three capture modes (fullscreen / window / region) on the sender side, with platform-specific capture backends, region selection overlay, and proper UI state machine transitions.

**Architecture:** Expand `CaptureMode` enum with Window/Region variants, add `WindowInfo` + `CaptureError`, modify `VideoCapture` trait to return `Result<Option<CapturedFrame>, CaptureError>`, wire the UI with a region selection overlay reusing egui's event loop, and adapt negotiation to derive resolution from the selected mode.

**Tech Stack:** Rust, eframe/egui 0.34, ScreenCaptureKit (macOS), DXGI/WGC (Windows), egui-toast

```rust
use crate::ui::messages::WindowInfo;

pub fn list_windows_macos() -> Vec<WindowInfo> {
    let Ok(content) = SCShareableContent::get() else {
        return vec![];
    };
    content
        .windows()
        .into_iter()
        .filter(|w| w.is_on_screen() && w.window_layer() == 0)
        .filter_map(|w| {
            let title = w.title().unwrap_or_default();
            let app_name = w.owning_application()
                .map(|a| a.application_name().unwrap_or_default())
                .unwrap_or_default();
            if title.is_empty() && app_name.is_empty() {
                return None;
            }
            Some(WindowInfo {
                id: w.window_id() as u64,
                title,
                app_name,
            })
        })
        .collect()
}
```

```rust
#[cfg(target_os = "macos")]
fn take_screenshot_macos() -> Result<Screenshot> {
    type CGImageRef = *mut std::ffi::c_void;
    type CFDataRef = *mut std::ffi::c_void;

    #[link(name = "CoreGraphics", kind = "framework")]
    unsafe extern "C" {
        fn CGMainDisplayID() -> u32;
        fn CGDisplayCreateImage(display_id: u32) -> CGImageRef;
        fn CGImageGetWidth(image: CGImageRef) -> usize;
        fn CGImageGetHeight(image: CGImageRef) -> usize;
        fn CGImageRelease(image: CGImageRef);
    }

    #[link(name = "CoreGraphics", kind = "framework")]
    unsafe extern "C" {
        fn CGDataProviderCopyData(provider: *mut std::ffi::c_void) -> CFDataRef;
        fn CGImageGetDataProvider(image: CGImageRef) -> *mut std::ffi::c_void;
        fn CGImageGetBitsPerPixel(image: CGImageRef) -> usize;
        fn CGImageGetBytesPerRow(image: CGImageRef) -> usize;
    }

    #[link(name = "CoreFoundation", kind = "framework")]
    unsafe extern "C" {
        fn CFDataGetBytePtr(data: CFDataRef) -> *const u8;
        fn CFDataGetLength(data: CFDataRef) -> isize;
        fn CFRelease(cf: *mut std::ffi::c_void);
    }

    unsafe {
        let display_id = CGMainDisplayID();
        let image = CGDisplayCreateImage(display_id);
        if image.is_null() {
            anyhow::bail!("CGDisplayCreateImage returned null");
        }

        let width = CGImageGetWidth(image) as u32;
        let height = CGImageGetHeight(image) as u32;
        let bpp = CGImageGetBitsPerPixel(image);
        let bytes_per_row = CGImageGetBytesPerRow(image);

        let provider = CGImageGetDataProvider(image);
        let data = CGDataProviderCopyData(provider);
        if data.is_null() {
            CGImageRelease(image);
            anyhow::bail!("CGDataProviderCopyData returned null");
        }

        let ptr = CFDataGetBytePtr(data);
        let len = CFDataGetLength(data) as usize;
        let raw_bytes = std::slice::from_raw_parts(ptr, len);

        // CGDisplayCreateImage returns BGRA (32-bit), convert to RGBA
        let mut rgba = Vec::with_capacity((width * height * 4) as usize);
        for row in 0..height as usize {
            let row_start = row * bytes_per_row;
            for col in 0..width as usize {
                let offset = row_start + col * (bpp / 8);
                if offset + 3 < len {
                    rgba.push(raw_bytes[offset + 2]); // R (from B)
                    rgba.push(raw_bytes[offset + 1]); // G
                    rgba.push(raw_bytes[offset]);     // B (from R)
                    rgba.push(raw_bytes[offset + 3]); // A
                }
            }
        }

        CFRelease(data);
        CGImageRelease(image);

        Ok(Screenshot { width, height, rgba })
    }
}
```

```rust
use eframe::egui::{self, Color32, Pos2, Rect, Sense, Stroke, Vec2};

pub struct SelectedRegion {
    pub x: f64,
    pub y: f64,
    pub width: f64,
    pub height: f64,
}

enum DragState {
    Idle,
    Dragging { start: Pos2, current: Pos2 },
    Selected { rect: Rect },
}

pub enum RegionAction {
    None,
    Confirmed(SelectedRegion),
    Cancelled,
}

pub struct RegionSelectView {
    texture: Option<egui::TextureHandle>,
    image_size: [u32; 2],
    state: DragState,
}

impl RegionSelectView {
    pub fn new(rgba: Vec<u8>, width: u32, height: u32) -> Self {
        Self {
            texture: None,
            image_size: [width, height],
            state: DragState::Idle,
        }
    }

    fn ensure_texture(&mut self, ctx: &egui::Context, rgba: &[u8]) {
        if self.texture.is_none() {
            let image = egui::ColorImage::from_rgba_unmultiplied(
                [self.image_size[0] as usize, self.image_size[1] as usize],
                rgba,
            );
            self.texture = Some(ctx.load_texture("screenshot", image, egui::TextureOptions::LINEAR));
        }
    }

    pub fn render(&mut self, ui: &mut egui::Ui, rgba: &[u8]) -> RegionAction {
        self.ensure_texture(ui.ctx(), rgba);

        let available = ui.available_size();
        let (rect, response) = ui.allocate_exact_size(available, Sense::click_and_drag());

        // Draw screenshot
        if let Some(tex) = &self.texture {
            let uv = Rect::from_min_max(Pos2::ZERO, Pos2::new(1.0, 1.0));
            ui.painter().image(tex.id(), rect, uv, Color32::WHITE);
        }

        // Dark overlay
        ui.painter().rect_filled(rect, 0.0, Color32::from_black_alpha(100));

        // Handle input
        if ui.input(|i| i.key_pressed(egui::Key::Escape)) {
            return RegionAction::Cancelled;
        }

        if ui.input(|i| i.key_pressed(egui::Key::Enter)) {
            if let DragState::Selected { rect: sel } = &self.state {
                let scale_x = self.image_size[0] as f64 / available.x as f64;
                let scale_y = self.image_size[1] as f64 / available.y as f64;
                return RegionAction::Confirmed(SelectedRegion {
                    x: (sel.min.x - rect.min.x) as f64 * scale_x,
                    y: (sel.min.y - rect.min.y) as f64 * scale_y,
                    width: sel.width() as f64 * scale_x,
                    height: sel.height() as f64 * scale_y,
                });
            }
        }

        // Drag handling
        if response.drag_started() {
            if let Some(pos) = response.interact_pointer_pos() {
                self.state = DragState::Dragging { start: pos, current: pos };
            }
        }

        if response.dragged() {
            if let DragState::Dragging { start, ref mut current } = &mut self.state {
                if let Some(pos) = response.interact_pointer_pos() {
                    *current = pos;
                }
            }
        }

        if response.drag_stopped() {
            if let DragState::Dragging { start, current } = &self.state {
                let sel = Rect::from_two_pos(*start, *current);
                if sel.width() > 10.0 && sel.height() > 10.0 {
                    self.state = DragState::Selected { rect: sel };
                } else {
                    self.state = DragState::Idle;
                }
            }
        }

        // Draw selection rectangle
        match &self.state {
            DragState::Dragging { start, current } => {
                let sel = Rect::from_two_pos(*start, *current);
                // Clear overlay in selection area (draw screenshot again)
                if let Some(tex) = &self.texture {
                    let uv_min = Pos2::new(
                        (sel.min.x - rect.min.x) / rect.width(),
                        (sel.min.y - rect.min.y) / rect.height(),
                    );
                    let uv_max = Pos2::new(
                        (sel.max.x - rect.min.x) / rect.width(),
                        (sel.max.y - rect.min.y) / rect.height(),
                    );
                    ui.painter().image(tex.id(), sel, Rect::from_min_max(uv_min, uv_max), Color32::WHITE);
                }
                ui.painter().rect_stroke(sel, 0.0, Stroke::new(2.0, Color32::from_rgb(59, 130, 246)));
                // Size label
                let scale_x = self.image_size[0] as f32 / available.x;
                let scale_y = self.image_size[1] as f32 / available.y;
                let label = format!(
                    "{}x{}",
                    (sel.width() * scale_x) as u32,
                    (sel.height() * scale_y) as u32
                );
                ui.painter().text(
                    sel.right_bottom() + Vec2::new(4.0, 4.0),
                    egui::Align2::LEFT_TOP,
                    label,
                    egui::FontId::proportional(12.0),
                    Color32::WHITE,
                );
            }
            DragState::Selected { rect: sel } => {
                if let Some(tex) = &self.texture {
                    let uv_min = Pos2::new(
                        (sel.min.x - rect.min.x) / rect.width(),
                        (sel.min.y - rect.min.y) / rect.height(),
                    );
                    let uv_max = Pos2::new(
                        (sel.max.x - rect.min.x) / rect.width(),
                        (sel.max.y - rect.min.y) / rect.height(),
                    );
                    ui.painter().image(tex.id(), *sel, Rect::from_min_max(uv_min, uv_max), Color32::WHITE);
                }
                ui.painter().rect_stroke(*sel, 0.0, Stroke::new(2.0, Color32::from_rgb(34, 197, 94)));
                let scale_x = self.image_size[0] as f32 / available.x;
                let scale_y = self.image_size[1] as f32 / available.y;
                let label = format!(
                    "{}x{} — Enter确认 / Esc取消",
                    (sel.width() * scale_x) as u32,
                    (sel.height() * scale_y) as u32
                );
                ui.painter().text(
                    sel.right_bottom() + Vec2::new(4.0, 4.0),
                    egui::Align2::LEFT_TOP,
                    label,
                    egui::FontId::proportional(12.0),
                    Color32::WHITE,
                );
            }
            DragState::Idle => {}
        }

        RegionAction::None
    }
}
```

```rust
impl VideoCapture for MacOsCapture {
    fn new(config: &CaptureConfig) -> anyhow::Result<Self> {
        let content = SCShareableContent::get()?;
        let display = content
            .displays()
            .into_iter()
            .next()
            .ok_or_else(|| anyhow::anyhow!("no display found"))?;

        let (cap_w, cap_h) = if config.width == 0 || config.height == 0 {
            let display_id = display.display_id();
            physical_display_size(display_id)
                .or_else(|| {
                    let fallback_id = unsafe { CGMainDisplayID() };
                    physical_display_size(fallback_id)
                })
                .unwrap_or((1920, 1080))
        } else {
            (config.width, config.height)
        };

        let filter = match &config.mode {
            CaptureMode::FullScreen => {
                SCContentFilter::create()
                    .with_display(&display)
                    .with_excluding_windows(&[])
                    .build()
            }
            CaptureMode::Window { id, .. } => {
                let window = content
                    .windows()
                    .into_iter()
                    .find(|w| w.window_id() as u64 == *id)
                    .ok_or_else(|| anyhow::anyhow!("window not found: {id}"))?;
                SCContentFilter::create()
                    .with_window(&window)
                    .build()
            }
            CaptureMode::Region { .. } => {
                SCContentFilter::create()
                    .with_display(&display)
                    .with_excluding_windows(&[])
                    .build()
            }
        };

        let mut sc_config = SCStreamConfiguration::new()
            .with_width(cap_w)
            .with_height(cap_h)
            .with_fps(config.fps)
            .with_captures_audio(true)
            .with_sample_rate(48000)
            .with_channel_count(2);

        // Apply source_rect for Region mode (hardware crop)
        if let CaptureMode::Region { x, y, width, height } = &config.mode {
            sc_config = sc_config.with_source_rect(screencapturekit::prelude::CGRect {
                origin: screencapturekit::prelude::CGPoint { x: *x, y: *y },
                size: screencapturekit::prelude::CGSize { width: *width, height: *height },
            });
        }

        let (video_tx, video_rx) = mpsc::sync_channel(2);
        let (audio_tx, audio_rx) = mpsc::sync_channel(8);

        let mut stream = SCStream::new(&filter, &sc_config);
        stream
            .add_output_handler(VideoHandler { tx: video_tx }, SCStreamOutputType::Screen)
            .ok_or_else(|| anyhow::anyhow!("failed to add video output handler"))?;
        stream
            .add_output_handler(AudioHandler { tx: audio_tx }, SCStreamOutputType::Audio)
            .ok_or_else(|| anyhow::anyhow!("failed to add audio output handler"))?;
        stream.start_capture()?;

        Ok(Self {
            _stream: stream,
            video_rx,
            audio_rx,
            width: cap_w,
            height: cap_h,
        })
    }
}
```
